import { NextResponse } from "next/server";
import { Pool } from "pg";
import { randomUUID } from "node:crypto";

export const runtime = "nodejs";

const globalForPool = globalThis as unknown as {
  ordersPool?: Pool;
};

const pool =
  globalForPool.ordersPool ??
  new Pool({
    connectionString: process.env.DATABASE_URL,
    ssl: { rejectUnauthorized: false },
    max: 2,
  });

if (process.env.NODE_ENV !== "production") {
  globalForPool.ordersPool = pool;
}

export async function POST(request: Request) {
  let body: {
    username?: unknown;
    robuxAmount?: unknown;
    buyerEmail?: unknown;
    buyerWhatsapp?: unknown;
  };

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "Format permintaan tidak valid." },
      { status: 400 }
    );
  }

  const username =
    typeof body.username === "string" ? body.username.trim() : "";
  const amount = Number(body.robuxAmount);

  if (!/^[A-Za-z0-9_]{3,20}$/.test(username)) {
    return NextResponse.json(
      { error: "Username Roblox tidak valid." },
      { status: 400 }
    );
  }

  if (
    !Number.isSafeInteger(amount) ||
    amount < 10 ||
    amount > 10000
  ) {
    return NextResponse.json(
      { error: "Jumlah Robux harus 10–10.000." },
      { status: 400 }
    );
  }

  const buyerEmail =
    typeof body.buyerEmail === "string" ? body.buyerEmail.trim() : null;
  const buyerWhatsapp =
    typeof body.buyerWhatsapp === "string"
      ? body.buyerWhatsapp.trim()
      : null;

  if (buyerEmail && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(buyerEmail)) {
    return NextResponse.json(
      { error: "Format email tidak valid." },
      { status: 400 }
    );
  }

  if (buyerWhatsapp && !/^[0-9+()\-\s]{8,20}$/.test(buyerWhatsapp)) {
    return NextResponse.json(
      { error: "Nomor WhatsApp tidak valid." },
      { status: 400 }
    );
  }

  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const config = await client.query(
      'SELECT "totalStock", "reservedStock", "pricePerRobux" FROM "StoreConfig" WHERE id = 1 FOR UPDATE'
    );

    if (!config.rows.length) {
      await client.query("ROLLBACK");
      return NextResponse.json(
        { error: "Konfigurasi stok belum tersedia." },
        { status: 503 }
      );
    }

    const row = config.rows[0];
    const availableStock =
      Number(row.totalStock) - Number(row.reservedStock);
    const pricePerRobux = Number(row.pricePerRobux);

    if (amount > availableStock) {
      await client.query("ROLLBACK");
      return NextResponse.json(
        { error: "Stok Robux tidak mencukupi.", availableStock },
        { status: 409 }
      );
    }

    // Harga dihitung dari database, tidak dari input pengguna.
    const totalPrice = amount * pricePerRobux;

    if (!Number.isSafeInteger(totalPrice)) {
      await client.query("ROLLBACK");
      return NextResponse.json(
        { error: "Total harga tidak valid." },
        { status: 400 }
      );
    }

    const orderId = randomUUID();

    await client.query(
      `INSERT INTO "Order"
        (id, "robloxUsername", "robuxAmount", "pricePerRobux",
         "totalPrice", "buyerEmail", "buyerWhatsapp", status,
         "createdAt", "updatedAt", "expiresAt")
       VALUES ($1, $2, $3, $4, $5, $6, $7, 'PENDING',
               NOW(), NOW(), NOW() + INTERVAL '30 minutes')`,
      [
        orderId,
        username,
        amount,
        pricePerRobux,
        totalPrice,
        buyerEmail,
        buyerWhatsapp,
      ]
    );

    await client.query(
    `UPDATE "StoreConfig"
     SET "reservedStock" = "reservedStock" + $1,
         "updatedAt" = NOW()
     WHERE id = 1`,
    [amount]
  );

  await client.query("COMMIT");

    return NextResponse.json(
      {
        orderId,
        username,
        robuxAmount: amount,
        pricePerRobux,
        totalPrice,
        currency: "IDR",
        status: "PENDING",
        expiresAt: new Date(Date.now() + 30 * 60 * 1000).toISOString(),
      },
      { status: 201 }
    );
  } catch (error) {
    await client.query("ROLLBACK").catch(() => {});
    console.error("Create order error:", error);
    return NextResponse.json(
      { error: "Pesanan gagal dibuat. Periksa struktur database." },
      { status: 500 }
    );
  } finally {
    client.release();
  }
}

import { NextResponse } from "next/server";
import { Pool } from "pg";

export const runtime = "nodejs";

const globalForPool = globalThis as unknown as {
  paymentPool?: Pool;
};

const pool =
  globalForPool.paymentPool ??
  new Pool({
    connectionString: process.env.DATABASE_URL,
    ssl: { rejectUnauthorized: false },
    max: 2,
  });

if (process.env.NODE_ENV !== "production") {
  globalForPool.paymentPool = pool;
}

export async function POST(request: Request) {
  const apiKey = process.env.KASERA_PAY_KEY;

  if (!apiKey || !apiKey.startsWith("kp_test_")) {
    return NextResponse.json(
      { error: "API key test Kasera Pay belum dikonfigurasi." },
      { status: 503 }
    );
  }

  let body: { orderId?: unknown };

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "Permintaan tidak valid." },
      { status: 400 }
    );
  }

  if (
    typeof body.orderId !== "string" ||
    body.orderId.length > 100
  ) {
    return NextResponse.json(
      { error: "ID pesanan tidak valid." },
      { status: 400 }
    );
  }

  let order;

  try {
    const result = await pool.query(
      `SELECT id, "robuxAmount", "totalPrice", status, "expiresAt"
       FROM "Order"
       WHERE id = $1
       LIMIT 1`,
      [body.orderId]
    );

    order = result.rows[0];
  } catch (error) {
    console.error("Read order for checkout failed:", error);
    return NextResponse.json(
      { error: "Gagal memeriksa pesanan." },
      { status: 500 }
    );
  }

  if (!order) {
    return NextResponse.json(
      { error: "Pesanan tidak ditemukan." },
      { status: 404 }
    );
  }

  if (
    order.status !== "PENDING" ||
    !order.expiresAt ||
    new Date(order.expiresAt).getTime() <= Date.now()
  ) {
    return NextResponse.json(
      { error: "Pesanan sudah tidak aktif atau telah kedaluwarsa." },
      { status: 409 }
    );
  }

  const amount = Number(order.totalPrice);
  const robuxAmount = Number(order.robuxAmount);

  if (
    !Number.isSafeInteger(amount) ||
    amount < 1000 ||
    !Number.isSafeInteger(robuxAmount) ||
    robuxAmount < 10 ||
    robuxAmount > 10000
  ) {
    return NextResponse.json(
      { error: "Nominal pesanan tidak valid." },
      { status: 400 }
    );
  }

  const payload = {
    amount,
    description: `Robux Store - ${robuxAmount} Robux`,
    external_id: order.id,
    merchant_ref: order.id,
    payment_methods: ["qris"],
    return_url: "https://robux-store-ashen.vercel.app/",
    expires_in_minutes: 30,
  };

  let response: Response;

  try {
    response = await fetch(
      "https://pay.kasera.id/v1/transactions",
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${apiKey}`,
          "Content-Type": "application/json",
          "Idempotency-Key": order.id,
        },
        body: JSON.stringify(payload),
        cache: "no-store",
        signal: AbortSignal.timeout(15000),
      }
    );
  } catch (error) {
    console.error("Kasera checkout request failed:", error);
    return NextResponse.json(
      { error: "Kasera Pay tidak dapat dihubungi. Coba lagi nanti." },
      { status: 502 }
    );
  }

  const data = await response.json().catch(() => null);

  if (!response.ok) {
    console.error("Kasera checkout rejected:", response.status);
    return NextResponse.json(
      { error: "Kasera Pay menolak permintaan checkout." },
      { status: 502 }
    );
  }

  if (
    data?.livemode !== false ||
    typeof data?.checkout_url !== "string" ||
    typeof data?.id !== "string"
  ) {
    console.error("Unexpected Kasera test response.");
    return NextResponse.json(
      { error: "Respons test Kasera Pay tidak sesuai." },
      { status: 502 }
    );
  }

  let checkoutUrl: URL;

  try {
    checkoutUrl = new URL(data.checkout_url);
  } catch {
    return NextResponse.json(
      { error: "Link checkout tidak valid." },
      { status: 502 }
    );
  }

  if (checkoutUrl.origin !== "https://pay.kasera.id") {
    return NextResponse.json(
      { error: "Domain checkout tidak diizinkan." },
      { status: 502 }
    );
  }

  return NextResponse.json({
    orderId: order.id,
    status: "PENDING",
    checkoutUrl: checkoutUrl.toString(),
    expiresAt: data.expires_at ?? null,
  });
}

===== src/app/page.tsx =====
"use client";

import { useEffect, useState } from "react";

type Config = {
  stock: number;
  pricePerRobux: number;
  currency: string;
};

type RobloxUser = {
  id: number;
  username: string;
  displayName: string;
  avatar: string | null;
};

const money = (n: number) =>
  new Intl.NumberFormat("id-ID", {
    style: "currency",
    currency: "IDR",
    maximumFractionDigits: 0,
  }).format(n);

export default function Home() {
  const [config, setConfig] = useState<Config | null>(null);
  const [username, setUsername] = useState("");
  const [amount, setAmount] = useState("100");
  const [user, setUser] = useState<RobloxUser | null>(null);
  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);
  const [checkoutLoading, setCheckoutLoading] = useState(false);

  useEffect(() => {
    let active = true;

    const load = async () => {
      try {
        const r = await fetch("/api/config", { cache: "no-store" });
        if (!r.ok) throw new Error();
        const data = await r.json();
        if (active) setConfig(data);
      } catch {
        if (active) setMessage("Stok belum dapat dimuat dari server.");
      }
    };

    void load();
    const timer = setInterval(load, 15000);

    return () => {
      active = false;
      clearInterval(timer);
    };
  }, []);

  const quantity = Number(amount);
  const valid =
    Number.isSafeInteger(quantity) &&
    quantity >= 10 &&
    quantity <= (config?.stock ?? 0);

  const total =
    config && valid ? quantity * config.pricePerRobux : 0;

  async function checkout() {
    if (!user || !config || !valid || checkoutLoading) return;

    setCheckoutLoading(true);
    setMessage("");

    try {
      const orderResponse = await fetch("/api/orders", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          username: user.username,
          robuxAmount: quantity,
        }),
      });

      const orderData = await orderResponse.json();
      if (!orderResponse.ok) {
        throw new Error(orderData.error || "Gagal membuat pesanan.");
      }

      const orderId = orderData.order?.id || orderData.orderId || orderData.id;
      if (!orderId) {
        throw new Error("Pesanan dibuat, tetapi ID pesanan tidak ditemukan. Periksa respons API sebelum mencoba lagi.");
      }

      const paymentResponse = await fetch("/api/payments/create", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ orderId }),
      });

      const paymentData = await paymentResponse.json();
      if (!paymentResponse.ok) {
        throw new Error(paymentData.error || "Gagal membuat checkout pembayaran.");
      }

      const checkoutUrl = paymentData.checkoutUrl || paymentData.checkout_url;
      if (typeof checkoutUrl !== "string" || !checkoutUrl) {
        throw new Error("Tautan checkout tidak ditemukan.");
      }

      window.location.href = checkoutUrl;
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "Checkout gagal.");
    } finally {
      setCheckoutLoading(false);
    }
  }

async function checkUser() {
    setLoading(true);
    setUser(null);
    setMessage("");

    try {
      const r = await fetch("/api/roblox", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ username: username.trim() }),
      });

      const data = await r.json();
      if (!r.ok) throw new Error(data.error || "Username tidak ditemukan.");

      setUser(data);
      setMessage("Username ditemukan. Periksa akun tujuan.");
    } catch (e) {
      setMessage(e instanceof Error ? e.message : "Terjadi kesalahan.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <main className="min-h-screen bg-[#090b12] text-white">
      <header className="border-b border-white/10">
        <div className="mx-auto flex max-w-5xl items-center justify-between px-5 py-5">
          <div className="flex items-center gap-3">
            <div className="grid h-10 w-10 place-items-center rounded-xl bg-violet-600 font-black">R</div>
            <div>
              <p className="font-bold">ROBUX STORE</p>
              <p className="text-xs text-slate-400">Top up praktis</p>
            </div>
          </div>
          <span className="text-xs text-emerald-300">● Toko online</span>
        </div>
      </header>

      <section className="mx-auto max-w-5xl px-5 py-12">
        <p className="mb-3 text-sm font-bold tracking-[.2em] text-violet-400">
          ROBLOX TOP UP
        </p>
        <h1 className="text-4xl font-black leading-tight md:text-6xl">
          Robux untuk
          <span className="block text-violet-400">petualanganmu.</span>
        </h1>
        <p className="mt-4 max-w-xl text-slate-400">
          Masukkan username Roblox, pilih jumlah, dan periksa estimasi harga.
        </p>

        <div className="mt-9 grid gap-6 lg:grid-cols-[1fr_320px]">
          <section className="rounded-3xl border border-white/10 bg-[#111521] p-5 md:p-8">
            <h2 className="text-xl font-bold">Buat pesanan</h2>

            <label className="mb-2 mt-6 block text-sm">Username Roblox</label>
            <div className="flex flex-col gap-2 sm:flex-row">
              <input
                value={username}
                onChange={(e) => {
                  setUsername(e.target.value);
                  setUser(null);
                }}
                placeholder="Username Roblox"
                maxLength={20}
                className="min-w-0 flex-1 rounded-xl border border-white/10 bg-[#090d17] px-4 py-3 outline-none focus:border-violet-500"
              />
              <button
                onClick={checkUser}
                disabled={loading || username.trim().length < 3}
                className="rounded-xl bg-white/10 px-5 py-3 font-bold disabled:opacity-40"
              >
                {loading ? "Memeriksa..." : "Periksa"}
              </button>
            </div>

            {user && (
              <div className="mt-4 flex items-center gap-3 rounded-xl border border-emerald-400/20 p-3">
                {user.avatar && (
                  <img src={user.avatar} alt="Avatar Roblox" className="h-12 w-12 rounded-lg" />
                )}
                <div>
                  <p className="font-bold">{user.displayName}</p>
                  <p className="text-sm text-slate-400">@{user.username}</p>
                </div>
              </div>
            )}

            <label className="mb-2 mt-7 block text-sm">Jumlah Robux</label>
            <input
              type="number"
              min={10}
              max={config?.stock ?? 0}
              step={1}
              value={amount}
              onChange={(e) => setAmount(e.target.value)}
              className="w-full rounded-xl border border-white/10 bg-[#090d17] px-4 py-4 text-lg font-bold outline-none focus:border-violet-500"
            />

            <div className="mt-3 flex flex-wrap gap-2">
              {[100, 250, 500, 1000].map((n) => (
                <button
                  key={n}
                  disabled={!config || n > config.stock}
                  onClick={() => setAmount(String(n))}
                  className="rounded-lg border border-white/10 px-3 py-2 text-sm disabled:opacity-30"
                >
                  {n.toLocaleString("id-ID")}
                </button>
              ))}
            </div>

            <div className="mt-7 rounded-2xl border border-violet-400/20 bg-violet-400/5 p-5">
              <div className="flex justify-between gap-3 text-sm text-slate-400">
                <span>Harga per Robux</span>
                <span className="text-white">
                  {config ? money(config.pricePerRobux) : "Memuat..."}
                </span>
              </div>
              <div className="mt-4 flex items-end justify-between gap-3">
                <span className="text-sm text-slate-400">Estimasi total</span>
                <span className="text-2xl font-black text-violet-300">
                  {config ? money(total) : "—"}
                </span>
              </div>
            </div>

            {message && <p role="status" className="mt-4 text-sm text-slate-300">{message}</p>}

            <button
              disabled={!config || config.stock <= 0 || !valid || !user || checkoutLoading}
              onClick={checkout}
              className="mt-5 w-full rounded-xl bg-violet-600 px-5 py-4 font-bold disabled:cursor-not-allowed disabled:bg-slate-700"
            >
              {checkoutLoading ? "Menyiapkan pembayaran..." : !config ? "Memuat toko..." : config.stock <= 0 ? "Stok habis" : "Lanjut ke pembayaran"}
            </button>
          </section>

          <aside className="rounded-3xl border border-white/10 bg-[#111521] p-6">
            <p className="text-sm text-slate-400">Sisa limit Robux</p>
            <p className="mt-3 text-4xl font-black">
              {config ? config.stock.toLocaleString("id-ID") : "—"}
            </p>
            <p className="mt-1 text-sm text-slate-400">Robux tersedia</p>
            {config?.stock === 0 && (
              <p className="mt-4 text-sm font-bold text-red-300">Stok habis</p>
            )}
            <hr className="my-6 border-white/10" />
            <h3 className="font-bold">Cara pembelian</h3>
            <ol className="mt-4 space-y-4 text-sm text-slate-400">
              <li>01 — Periksa username Roblox.</li>
              <li>02 — Tentukan jumlah Robux.</li>
              <li>03 — Periksa estimasi harga.</li>
            </ol>
            <p className="mt-6 text-xs leading-5 text-slate-500">
              Jangan pernah memberikan password Roblox kepada toko.
              Pembayaran belum aktif pada Fase 1.
            </p>
          </aside>
        </div>
      </section>

      <footer className="border-t border-white/10 px-5 py-7 text-center text-xs text-slate-500">
        © {new Date().getFullYear()} Robux Store
      </footer>
    </main>
  );
}
